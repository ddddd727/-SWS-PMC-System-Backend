## 🔧 统一响应格式使用指南
### 1. 后端API开发规范
#### 1.1 控制器必须继承基类
```csharp
// ✅ 正确做法
[ApiController]
[Route("api/[controller]")]
public class UserController : ApiControllerBase  // 必须继承基类
{
    private readonly IUserService _userService;
    
    public UserController(IUserService userService)
    {
        _userService = userService;
    }
    
    [HttpGet("{id}")]
    public async Task<IActionResult> GetUser(int id)
    {
        // 业务逻辑...
    }
}

// ❌ 错误做法
public class UserController : ControllerBase  // 缺少统一基类
{
    // 这将导致响应格式不一致
}
```

#### 1.2 标准响应方法使用
| 方法 | 用途 | HTTP状态码 | 示例 |
| --- | --- | --- | --- |
| `Success<T>(data)` | 成功返回数据 | 200 | `return Success(user);` |
| `Success(message)` | 成功无数据 | 200 | `return Success("操作成功");` |
| `Created<T>(data)` | 创建成功 | 201 | `return Created(newUser);` |
| `Paged<T>(result)` | 分页数据 | 200 | `return Paged(pagedResult);` |
| `Fail(errorCode)` | 业务失败 | 400/404等 | `return Fail(ApiErrorCode.NotFound);` |
| `ValidationFailed()` | 验证失败 | 400 | `return ValidationFailed();` |


#### 1.3 完整API示例
```csharp
/// <summary>
/// 用户管理API
/// </summary>
[ApiController]
[Route("api/[controller]")]
[Authorize]  // 如需权限控制
public class UserController : ApiControllerBase
{
    private readonly IUserService _userService;
    
    public UserController(IUserService userService)
    {
        _userService = userService;
    }
    
    /// <summary>
    /// 获取用户列表（分页）
    /// </summary>
    /// <param name="request">分页参数</param>
    /// <response code="200">返回用户列表</response>
    /// <response code="400">参数验证失败</response>
    [HttpGet]
    [ProducesResponseType(typeof(ApiResponse<PagedResult<UserDto>>), 200)]
    [ProducesResponseType(typeof(ApiResponse), 400)]
    public async Task<IActionResult> GetUsers([FromQuery] PagedRequest request)
    {
        // 自动验证参数（如果request有DataAnnotation验证）
        if (AutoValidate() is IActionResult validationResult)
            return validationResult;
        
        var result = await _userService.GetPagedUsersAsync(request);
        return Paged(result);
    }
    
    /// <summary>
    /// 获取单个用户
    /// </summary>
    /// <param name="id">用户ID</param>
    /// <response code="200">返回用户信息</response>
    /// <response code="404">用户不存在</response>
    [HttpGet("{id:int}")]
    public async Task<IActionResult> GetUser(int id)
    {
        var user = await _userService.GetUserAsync(id);
        
        if (user == null)
            return Fail(ApiErrorCode.ResourceNotFound, $"用户ID {id} 不存在");
        
        return Success(user);
    }
    
    /// <summary>
    /// 创建用户
    /// </summary>
    /// <param name="dto">用户数据</param>
    [HttpPost]
    public async Task<IActionResult> CreateUser([FromBody] CreateUserDto dto)
    {
        // 手动验证DTO
        if (!ModelState.IsValid)
            return ValidationFailed();
        
        try
        {
            var createdUser = await _userService.CreateUserAsync(dto);
            return Created(createdUser);
        }
        catch (DuplicateException ex)
        {
            // 捕获业务异常，返回友好的错误信息
            return Fail(ApiErrorCode.DuplicateResource, ex.Message);
        }
    }
    
    /// <summary>
    /// 更新用户
    /// </summary>
    [HttpPut("{id:int}")]
    public async Task<IActionResult> UpdateUser(int id, [FromBody] UpdateUserDto dto)
    {
        if (!ModelState.IsValid)
            return ValidationFailed();
        
        var success = await _userService.UpdateUserAsync(id, dto);
        
        if (!success)
            return Fail(ApiErrorCode.ResourceNotFound);
        
        return Success("用户更新成功");
    }
    
    /// <summary>
    /// 删除用户
    /// </summary>
    [HttpDelete("{id:int}")]
    public async Task<IActionResult> DeleteUser(int id)
    {
        var success = await _userService.DeleteUserAsync(id);
        
        if (!success)
            return Fail(ApiErrorCode.ResourceNotFound);
        
        return Success("用户删除成功");
    }
    
    /// <summary>
    /// 批量操作示例
    /// </summary>
    [HttpPost("batch")]
    public async Task<IActionResult> BatchOperation([FromBody] BatchRequest request)
    {
        var result = await _userService.BatchOperationAsync(request);
        
        // 返回部分成功的结果
        return Success(new
        {
            Total = result.Total,
            SuccessCount = result.SuccessCount,
            FailedItems = result.FailedItems
        }, "批量操作完成");
    }
}
```

```csharp
public class UserService : IUserService

        var user = await _dbContext.Users.FindAsync(id);
```

### 3. EF Core数据库访问规范
#### 3.1 数据库上下文使用
```csharp
// 正确注入和使用DbContext
public class UserService : IUserService
{
    private readonly ApplicationDbContext _context;
    private readonly IMapper _mapper;
    
    public UserService(ApplicationDbContext context, IMapper mapper)
    {
        _context = context;
        _mapper = mapper;
    }
    
    public async Task<UserDto> GetUserAsync(int id)
    {
        // 使用异步方法
        var user = await _context.Users
            .AsNoTracking()  // 只读查询使用
            .FirstOrDefaultAsync(u => u.Id == id);
        
        return _mapper.Map<UserDto>(user);
    }
    
    public async Task<PagedResult<UserDto>> GetPagedUsersAsync(PagedRequest request)
    {
        // 分页查询
        var query = _context.Users.AsQueryable();
        
        // 条件过滤
        if (!string.IsNullOrEmpty(request.Keyword))
        {
            query = query.Where(u => u.Name.Contains(request.Keyword) ||
                                     u.Email.Contains(request.Keyword));
        }
        
        // 排序
        query = request.IsDescending 
            ? query.OrderByDescending(GetSortProperty(request.SortBy))
            : query.OrderBy(GetSortProperty(request.SortBy));
        
        // 获取总数
        var totalCount = await query.CountAsync();
        
        // 分页
        var items = await query
            .Skip((request.PageIndex - 1) * request.PageSize)
            .Take(request.PageSize)
            .AsNoTracking()
            .ToListAsync();
        
        return new PagedResult<UserDto>
        {
            Items = _mapper.Map<List<UserDto>>(items),
            TotalCount = totalCount,
            PageIndex = request.PageIndex,
            PageSize = request.PageSize
        };
    }
    
    public async Task<UserDto> CreateUserAsync(CreateUserDto dto)
    {
        // 检查重复
        var exists = await _context.Users
            .AnyAsync(u => u.Email == dto.Email);
        
        if (exists)
            throw new DuplicateException("邮箱已存在");
        
        // 创建实体
        var user = _mapper.Map<User>(dto);
        user.CreatedAt = DateTime.UtcNow;
        
        _context.Users.Add(user);
        await _context.SaveChangesAsync();
        
        return _mapper.Map<UserDto>(user);
    }
}
```

#### 3.2 性能优化要点
```csharp
// 1. 使用AsNoTracking()进行只读查询
var users = await _context.Users
    .AsNoTracking()  // 不跟踪变更，提高性能
    .Where(u => u.IsActive)
    .ToListAsync();

// 2. 避免N+1查询问题
// ❌ 错误做法（N+1查询）
var departments = await _context.Departments.ToListAsync();
foreach (var dept in departments)
{
    dept.Users = await _context.Users
        .Where(u => u.DepartmentId == dept.Id)
        .ToListAsync();
}

// ✅ 正确做法（使用Include或投影）
var departments = await _context.Departments
    .Include(d => d.Users)  // 使用Include加载关联数据
    .ToListAsync();

// 或使用投影
var result = await _context.Departments
    .Select(d => new 
    {
        Department = d,
        UserCount = d.Users.Count()
    })
    .ToListAsync();

// 3. 批量操作使用合适的方法
// ❌ 错误做法（逐个保存）
foreach (var item in items)
{
    _context.Add(item);
    await _context.SaveChangesAsync();  // 每次循环都保存
}

// ✅ 正确做法（批量保存）
foreach (var item in items)
{
    _context.Add(item);
}
await _context.SaveChangesAsync();  // 只保存一次

// 4. 使用事务
using var transaction = await _context.Database.BeginTransactionAsync();
try
{
    // 多个操作
    await _context.Users.AddAsync(user1);
    await _context.Users.AddAsync(user2);
    
    await _context.SaveChangesAsync();
    await transaction.CommitAsync();
}
catch
{
    await transaction.RollbackAsync();
    throw;
}
```

#### 4.3 统一错误处理
前端已经配置了全局错误拦截器，会自动处理以下情况：

1. **网络错误**：显示"网络连接失败"
2. **业务错误**：根据错误码显示对应消息
3. **401未授权**：自动跳转到登录页
4. **403禁止访问**：显示权限不足提示

如果需要特殊处理，可以在catch中实现：

```javascript
async function specialOperation() {
  try {
    await someApi.call();
  } catch (error) {
    // 全局拦截器已经显示了错误消息
    // 这里可以添加特殊逻辑
    if (error.message.includes('特定错误')) {
      // 执行特殊处理
      showCustomDialog();
    }
  }
}
```

### 5. 开发注意事项
#### 5.1 必做清单
- [ ] 所有控制器必须继承`ApiControllerBase`
- [ ] 使用`Success()`、`Fail()`等方法返回响应
- [ ] Service层抛出异常，不要返回错误响应
- [ ] 前端使用封装的API模块，不要直接使用axios
- [ ] 分页查询使用`PagedRequest`和`PagedResult`
- [ ] 数据库查询使用异步方法（`xxxAsync`）

#### 5.2 禁止行为
```csharp
// ❌ 禁止直接返回数据
[HttpGet("{id}")]
public User GetUser(int id)  // 错误：缺少统一包装
{
    return _userService.GetUser(id);
}

// ❌ 禁止手动包装响应
[HttpGet("{id}")]
public IActionResult GetUser(int id)
{
    var user = _userService.GetUser(id);
    
    // 错误：手动包装，可能导致格式不一致
    return Ok(new {
        success = true,
        data = user,
        time = DateTime.Now
    });
}

// ❌ 禁止在Service层使用HttpContext
public class UserService
{
    public void SomeMethod()
    {
        var context = new DefaultHttpContext();  // 错误！
    }
}
```

### 6. 常见问题解答
#### Q1: 如何返回自定义的错误消息？
```csharp
// 使用Fail方法并传入自定义消息
return Fail(ApiErrorCode.BusinessRuleViolation, "自定义错误消息");
```

#### Q2: 如何返回部分成功的结果？
```csharp
return Success(new {
    Total = 100,
    SuccessCount = 95,
    FailedCount = 5,
    FailedItems = failedList
}, "批量操作完成");
```

#### Q3: 前端如何获取HTTP状态码？
```javascript
// 正常情况下不需要，错误时会通过拦截器处理
// 如果需要特殊处理：
try {
  await api.someCall();
} catch (error) {
  if (error.response?.status === 404) {
    // 处理404
  }
}
```

#### Q4: 如何添加新的错误码？
1. 在`ApiErrorCode`枚举中添加新值
2. 在`ApiControllerBase.GetDefaultErrorMessage()`中添加默认消息
3. 在前端错误映射表中添加对应提示

#### Q5: 数据库迁移命令
```bash

# 更新数据库
dotnet ef database update

# 回滚迁移

# 删除迁移（未应用到数据库前）
dotnet ef migrations remove
```

### 7. 代码审查要点
审查API代码时，请检查以下内容：

| 检查项 | 通过标准 | 示例 |
| --- | --- | --- |
| 控制器继承 | 继承`ApiControllerBase` | ✅ `: ApiControllerBase` |
| 响应方法 | 使用基类方法 | ✅ `return Success(data);` |
| 异步方法 | 使用`async/await` | ✅ `public async Task<IActionResult>` |
| 异常处理 | Service层抛出，Controller层捕获 | ✅ `try-catch`在Controller |
| 参数验证 | 使用`AutoValidate()`或手动验证 | ✅ `if (!ModelState.IsValid)` |
| 注释文档 | 有XML注释和ProducesResponseType | ✅ `/// <summary>` |
| 分页查询 | 使用`PagedRequest/PagedResult` | ✅ `return Paged(result);` |


### 8. 紧急情况处理
#### 8.1 需要绕过统一格式的情况
极少数情况（如文件下载）需要直接返回结果：

```csharp

public IActionResult ExportData()

}
```
